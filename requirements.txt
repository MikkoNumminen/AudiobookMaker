# PDF parsing
PyMuPDF==1.24.5

# OCR fallback for image-only / scanned PDFs. ocrmypdf wraps Tesseract +
# Ghostscript into a single "add a searchable text layer to a PDF" pipeline.
# The Python package is imported lazily inside pdf_parser.py so the rest of
# the app stays importable when ocrmypdf is missing (dev installs without
# the OCR stack still work — they just can't read scanned PDFs).
# Frozen .exe builds bundle tesseract.exe + eng/fin language packs + gswin64
# alongside ffmpeg.exe; see audiobookmaker.spec.
#
# Pin policy: this is the version we end-to-end-validated against a 32-page
# scanned PDF during PR #26. ocrmypdf renamed a few public API entry points
# between the 16.x and 17.x lines, so a blind major-version bump risks
# breaking pdf_parser._run_ocr_fallback silently — re-validate before
# moving off 17.x. The pin gets refreshed alongside the other heavy deps
# (PyMuPDF, edge-tts) every 3-6 months.
ocrmypdf==17.4.2

# EPUB parsing
ebooklib==0.20
beautifulsoup4==4.14.3

# Text-to-speech
edge-tts==7.2.8

# Offline neural TTS (Piper). Voice models (~60MB each) are downloaded
# to ~/.audiobookmaker/piper_voices/ on first use, not bundled.
piper-tts==1.4.2

# Finnish text normalization
num2words==0.5.14

# Modern GUI toolkit

# Pillow — required by CustomTkinter for CTkImage (used by the hero
# header logo and button icons). CTk installs it transitively but we

# the PyInstaller spec's exclusion of PIL doesn't accidentally win.
Pillow==12.2.0

# Audio processing

# In-process audio playback for the Preview button (replaces shelling out

pyinstaller==6.8.0

# Dev-time secret loading. Scripts that need HF_TOKEN (voice-pack

# on os.environ — installed users never need secrets.
python-dotenv==1.0.1
